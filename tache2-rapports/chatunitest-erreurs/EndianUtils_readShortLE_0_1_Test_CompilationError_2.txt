import java.io.IOException;
import java.io.InputStream;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;
import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;

@ExtendWith(MockitoExtension.class)
public class EndianUtils_readShortLE_0_1_Test {

    @Test
    public void testReadShortLE() throws IOException, TikaException {
        InputStream inputStream = new ByteArrayInputStream(new byte[] { 0x01, 0x02 });
        short result = EndianUtils.readShortLE(inputStream);
        assertEquals(0x0201, result);
    }

    @Test
    public void testReadShortLEWithBufferUnderrun() throws IOException, TikaException {
        InputStream inputStream = new ByteArrayInputStream(new byte[] { 0x01 });
        assertThrows(BufferUnderrunException.class, () -> EndianUtils.readShortLE(inputStream));
    }

    @Test
    public void testReadShortLEWithIOException() throws IOException, TikaException {
        InputStream inputStream = mock(InputStream.class);
        when(inputStream.read()).thenReturn(-1);
        assertThrows(IOException.class, () -> EndianUtils.readShortLE(inputStream));
    }
}
