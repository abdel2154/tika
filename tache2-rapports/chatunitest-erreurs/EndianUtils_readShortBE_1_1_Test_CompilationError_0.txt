import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;

import org.mockito.junit.jupiter.MockitoExtension;
import java.io.InputStream;

public class EndianUtils_readShortBE_1_1_Test {

    @Test
    public void testReadShortBE() throws Exception {
        EndianUtils endianUtils = new EndianUtils();
        Method method = EndianUtils.class.getDeclaredMethod("readShortBE", InputStream.class);
        method.setAccessible(true);
        // Test normal case
        InputStream inputStream = new ByteArrayInputStream(new byte[] { 0x01, 0x02 });
        short result = (short) method.invoke(endianUtils, inputStream);
        assertEquals(0x0102, result);
        // Test BufferUnderrunException
        inputStream = new ByteArrayInputStream(new byte[] { 0x01 });
        Executable executable = () -> method.invoke(endianUtils, inputStream);
        assertThrows(BufferUnderrunException.class, executable);
    }
}
